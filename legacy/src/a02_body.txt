/* ==========================================================================
   THREE.js loader — tries several CDNs so the page works standalone.
   A UMD build is used deliberately: it keeps this a single file that also
   opens straight from disk (file://), where ES modules are blocked.
   ========================================================================== */
(function () {
  var SOURCES = [
    'https://unpkg.com/three@0.149.0/build/three.min.js',
    'https://cdn.jsdelivr.net/npm/three@0.149.0/build/three.min.js',
    'https://cdnjs.cloudflare.com/ajax/libs/three.js/0.149.0/three.min.js',
    'https://unpkg.com/three@0.147.0/build/three.min.js'
  ];
  var i = 0;
  function next() {
    if (i >= SOURCES.length) { fail(); return; }
    var s = document.createElement('script');
    s.src = SOURCES[i++];
    s.onload = function () { window.THREE ? start() : next(); };
    s.onerror = next;
    document.head.appendChild(s);
  }
  function fail() {
    document.getElementById('boot').style.display = 'none';
    var el = document.getElementById('nogl');
    el.querySelector('h2').textContent = 'Could not load the 3D library';
    el.querySelector('p').textContent =
      'Three.js could not be downloaded from any CDN. Check your internet connection and reload the page.';
    el.style.display = 'flex';
  }
  function start() {
    try {
      var c = document.createElement('canvas');
      if (!(c.getContext('webgl2') || c.getContext('webgl'))) throw 0;
    } catch (e) {
      document.getElementById('boot').style.display = 'none';
      document.getElementById('nogl').style.display = 'flex';
      return;
    }
    document.getElementById('boot-msg').textContent = 'Building your world';
    // The main script only DEFINES __GP_MAIN__; nothing touching THREE runs
    // until it is called here, once the library is guaranteed to be present.
    window.__GP_MAIN__();
  }
  window.__GP_START__ = next;
})();
</script>
